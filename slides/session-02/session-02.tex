\slide{
    \slidetitle{Acceptance Criteria}
    \vspace{0.5cm}
    {\large شرط‌هایی که می‌گویند یک User Story چه وقت «تمام» است.}
    
    \vspace{1cm}
    
    {\bfseries\textcolor{primary}{مثال برای US-01:}}
    \vspace{0.3cm}
    \begin{itemize}[itemsep=0.3em]
        \item صفحهٔ محصولات با ۲۰ آیتم نمایش داده شود
        \item هر آیتم شامل تصویر، نام، و قیمت باشد
        \item بارگذاری < ۲ ثانیه باشد
    \end{itemize}
}

    \vspace{0.5cm}

        \item {\large\textbf{سؤال:} الان سفارش‌ها را چطور ثبت می‌کنید؟}

        \item {\large\textbf{سؤال:} بزرگ‌ترین مشکل شما چیست؟}

        \item {\large\textbf{سؤال:} چه چیزی هرگز نباید اتفاق بیفتد؟}
        \item {\large\textbf{پاسخ:} سفارش اشتباه یا دوبار ثبت شود.}
    \end{itemize}
