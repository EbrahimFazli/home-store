\section{مقدمه}

\subsection{هدف سند}
این سند داستان‌های کاربر (User Stories) پروژهٔ هوم‌استور را تعریف می‌کند.

\subsection{قالب User Story}
هر User Story به این شکل نوشته می‌شود:

\begin{quote}
\textbf{به‌عنوان} [نقش]، \textbf{می‌خواهم} [کاری] \textbf{تا} [ارزش].
\end{quote}

\subsection{معیارهای INVEST}
\begin{itemize}
    \item \textbf{I}ndependent — مستقل
    \item \textbf{N}egotiable — قابل مذاکره
    \item \textbf{V}aluable — ارزشمند
    \item \textbf{E}stimable — قابل تخمین
    \item \textbf{S}mall — کوچک
    \item \textbf{T}estable — قابل تست
\end{itemize}

\subsection{اولویت‌بندی MoSCoW}
\begin{itemize}
    \item \textbf{Must:} بدون این، پروژه بی‌معنی است
    \item \textbf{Should:} مهم، اما اگر وقت نشد بعداً
    \item \textbf{Could:} خوب است داشته باشیم
    \item \textbf{Won't:} در این نسخه نداریم
\end{itemize}

\section{معیار پذیرش نمونه}

\subsection{US-01 — نمایش محصولات}

\textbf{به‌عنوان:} مشتری \\
\textbf{می‌خواهم:} محصولات را ببینم \\
\textbf{تا:} انتخاب کنم

\textbf{معیار پذیرش:}
\begin{itemize}
    \item صفحهٔ محصولات با حداقل ۲۰ آیتم نمایش داده شود
    \item هر آیتم شامل تصویر، نام، قیمت باشد
    \item بارگذاری صفحه < ۲ ثانیه باشد
    \item صفحه واکنش‌گرا باشد
\end{itemize}

\subsection{US-06 — افزودن به سبد خرید}

\textbf{به‌عنوان:} مشتری \\
\textbf{می‌خواهم:} محصول را به سبد اضافه کنم \\
\textbf{تا:} بعداً خرید کنم

\textbf{معیار پذیرش:}
\begin{itemize}
    \item دکمهٔ «افزودن به سبد» در صفحه باشد
    \item سبد بلافاصله به‌روز شود
    \item تعداد کل در هدر نمایش داده شود
    \item پیام موفقیت نمایش داده شود
\end{itemize}

\subsection{US-11 — ثبت سفارش}

\textbf{به‌عنوان:} مشتری \\
\textbf{می‌خواهم:} سفارش ثبت کنم \\
\textbf{تا:} محصول را دریافت کنم

\textbf{معیار پذیرش:}
\begin{itemize}
    \item فرم شامل نام، آدرس، تلفن باشد
    \item اعتبارسنجی همه فیلدها انجام شود
    \item بعد از ثبت، شماره پیگیری نمایش داده شود
\end{itemize}
