\section{Introduction}

\subsection{Purpose}
This Vision document defines the vision, scope, and stakeholders of the
\textbf{HomeStore} project. Its purpose is to establish a shared understanding
among the development team, the customer, and stakeholders regarding
\emph{what we are building, for whom, and why}.

This document is \textbf{not a formal contract}, but a \textbf{guiding reference}
for future decisions throughout the software lifecycle. It is reviewed
at the end of each Sprint.

\begin{itemize}
    \item \textbf{In scope:} Business goals, target users, MVP boundaries, success metrics, high-level risks.
    \item \textbf{Out of scope:} Technical details, UI design, detailed functional requirements (covered in the SRS).

\section{Product Vision}

\subsection{Vision Statement}

\begin{quote}
\textbf{For} owners of small home-appliance stores \textbf{who} sell authentic,
warranty-backed products but are only known to local customers,
\textbf{HomeStore} is an \textbf{online storefront}
\textbf{that} provides a simple, trustworthy catalog, comparison, and
ordering experience.

\textbf{Unlike} large marketplaces with multiple sellers and uncertain
quality, \textbf{our product} focuses on a single trusted seller with
genuine warranties.
\end{quote}

\subsection{Value Proposition}
\begin{center}
\begin{tabular}{|l|p{8cm}|}
\hline
\textbf{For} & \textbf{Value} \\
\hline
End customer & Fast purchase, easy comparison, warranty assurance \\
\hline
Store owner (Admin) & Access to customers beyond the neighborhood, simple product management \\
\hline
Warehouse staff & Accurate inventory tracking without paperwork \\
\hline
\end{tabular}
\end{center}

\subsection{Competitive Advantage}
\begin{itemize}
    \item \textbf{Simplicity:} A simpler UI than Digikala and Bamilo
    \item \textbf{Focus:} Only home appliances, nothing else
    \item \textbf{Trust:} Guaranteed authenticity and warranty
    \item \textbf{Local-first:} Support for customers in smaller cities
\end{itemize}

\section{Problem and Opportunity}

\subsection{Problem Statement}
Owners of small home-appliance stores face the following challenges:
\begin{enumerate}
    \item \textbf{Geographic limitation:} Only local customers buy from them.
    \item \textbf{No online channel:} They cannot offer products online.
    \item \textbf{Manual order process:} Orders are taken by phone, noted on paper, and error-prone.
    \item \textbf{Inventory opacity:} Customers do not know what is in stock.
    \item \textbf{No comparison ability:} Customers cannot compare products side by side.
\end{enumerate}

\subsection{Opportunity}
\begin{itemize}
    \item Growth of online home-appliance shopping in Iran
    \item Increasing internet penetration in small cities
    \item Absence of small stores in the online space
    \item Opportunity to create a niche market for authentic, warranty-backed products
\end{itemize}

\section{Product Scope}

\subsection{In Scope (MVP)}

\textbf{Functional:}
\begin{itemize}
    \item \textbf{F-01} Product catalog with images, prices, and specifications
    \item \textbf{F-02} Advanced search and filter (brand, price, energy rating, capacity)
    \item \textbf{F-03} Product detail page with warranty and technical specs
    \item \textbf{F-04} Compare 2--3 products in a table
    \item \textbf{F-05} Shopping cart (add, remove, change quantity)
    \item \textbf{F-06} Order submission with form validation
    \item \textbf{F-07} Admin panel: add/edit/delete products
    \item \textbf{F-08} Admin panel: view orders and update status
\end{itemize}

\textbf{Non-functional:}
\begin{itemize}
    \item \textbf{NF-01} Page load time < 2 seconds
    \item \textbf{NF-02} Support for 100 concurrent users
    \item \textbf{NF-03} Responsive UI for mobile and desktop
    \item \textbf{NF-04} Password hashing (bcrypt)
    \item \textbf{NF-05} Persian (Farsi) language support with RTL layout
\end{itemize}

\subsection{Out of Scope}
\begin{center}
\begin{tabular}{|l|p{8cm}|}
\hline
\textbf{Item} & \textbf{Reason} \\
\hline
Real payment gateway & Requires license and infrastructure --- simulation only \\
\hline
Mobile application & Web-only focus \\
\hline
Shipping and logistics & Beyond one semester \\
\hline
Real warranty and after-sales & Legal complexity \\
\hline
Online chat support & Low priority \\
\hline
Multilingual (Persian/English) & Persian-only focus \\
\hline
\end{tabular}
\end{center}

\subsection{Constraints}
\begin{itemize}
    \item \textbf{Time:} 14 weeks, 3 Sprints
    \item \textbf{Team:} 3--4 members
    \item \textbf{Technical:} No dedicated server; deploy on a free cloud service
    \item \textbf{Budget:} Zero (free tools only)
\end{itemize}

\subsection{Definition of Done}
A Story is ``Done'' when:
\begin{enumerate}
    \item The code is written and merged.
    \item Unit tests are written.
    \item Code review is completed.
    \item Related documentation is updated.
    \item It works in the demo environment.
\end{enumerate}
